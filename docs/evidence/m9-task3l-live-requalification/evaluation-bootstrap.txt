function l_schema($value,$schema){
 if(isset($schema['anyOf'])){foreach($schema['anyOf']as$branch)if(l_schema($value,$branch))return true;return false;}
 $type=$schema['type']??null;
 if($type==='object'){
  if(!is_array($value)||($value!==array()&&array_is_list($value)))return false;
  foreach($schema['required']??array()as$key)if(!array_key_exists($key,$value))return false;
  if(($schema['additionalProperties']??true)===false&&array_diff(array_keys($value),array_keys($schema['properties']??array())))return false;
  foreach($value as$key=>$item)if(isset($schema['properties'][$key])&&!l_schema($item,$schema['properties'][$key]))return false;
 }elseif($type==='array'){
  if(!is_array($value)||!array_is_list($value)||count($value)<($schema['minItems']??0)||count($value)>($schema['maxItems']??PHP_INT_MAX))return false;
  foreach($value as$item)if(isset($schema['items'])&&!l_schema($item,$schema['items']))return false;
 }elseif($type==='string'){if(!is_string($value)||strlen($value)<($schema['minLength']??0))return false;}
 elseif($type==='boolean'){if(!is_bool($value))return false;}
 elseif($type==='integer'){if(!is_int($value))return false;}
 elseif($type==='number'){if(!is_int($value)&&!is_float($value))return false;}
 elseif($type!==null)return false;
 return !isset($schema['enum'])||in_array($value,$schema['enum'],true);
}
